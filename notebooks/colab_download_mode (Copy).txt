    "import os, sys, subprocess, hashlib, time, json, tarfile, shutil\n",

    "import pandas as pd\n",

    "\n",

    "RAW_VENDOR_CSV = WORKSPACE_DIR / \"enamine_real_vendor_export.csv\"\n",
    "\n",
    "RAW_ARCHIVE_URL = \"https://huggingface.co/datasets/Yukk1Zz/if3-crossdocked2020/resolve/main/crossdocked_pocket10.tar.gz\"\n",

    "# URL for the authentic 130,000+ Enamine commercial building-block inventory (with real Z-catalog IDs)\n",
    "REAL_ENAMINE_URL = \"https://raw.githubusercontent.com/wenhao-gao/synnet-dataprep/main/data/building_blocks/enamine_bb_clean.csv\"\n",
    "\n",
    "def compute_sha256(path: Path) -> str:\n",
    "    hasher = hashlib.sha256()\n",
    "    with open(path, \"rb\") as f:\n",
    "        while chunk := f.read(4 * 1024 * 1024):\n",
    "            hasher.update(chunk)\n",
    "    return hasher.hexdigest()\n",
    "\n",
    "def is_valid_tar(path: Path) -> bool:\n",
    "    if not path.exists() or path.stat().st_size < 1e8:\n",
    "        return False\n",
    "    try:\n",
    "        with tarfile.open(path, \"r:gz\") as tar:\n",
    "            tar.next()\n",
    "        return True\n",
    "    except Exception:\n",
    "        return False\n",
    "\n",
    "print(\"=\" * 70)\n",
    "print(\"ACQUIRING 100% REAL-WORLD COMMERCIAL DATASETS\")\n",
    "print(\"=\" * 70)\n",
    "\n",
    "# ------------------------------------------------------------------------------\n",
    "# 1. Verify or Download crossdocked_pocket10.tar.gz (~1.6 GB)\n",
    "# ------------------------------------------------------------------------------\n",
    "print(\"\\n[1/3] Checking CrossDocked2020 raw archive (79,513 physical complexes)...\")\n",
    "if is_valid_tar(CROSSDOCKED_TAR):\n",
    "    print(f\"  -> Verified existing uncorrupted archive ({CROSSDOCKED_TAR.stat().st_size / 1e6:.1f} MB). Skipping download.\")\n",
    "else:\n",
    "    print(f\"  -> Downloading raw archive via aria2 from: {RAW_ARCHIVE_URL}\")\n",
    "    cmd = [\n",
    "        \"aria2c\", \"-x\", \"16\", \"-s\", \"16\", \"-k\", \"1M\", \"-c\",\n",
    "        \"-d\", str(WORKSPACE_DIR), \"-o\", \"crossdocked_pocket10.tar.gz\",\n",
    "        RAW_ARCHIVE_URL,\n",
    "    ]\n",
    "    subprocess.run(cmd, check=True)\n",
    "    if not is_valid_tar(CROSSDOCKED_TAR):\n",
    "        raise IOError(\"Downloaded CrossDocked archive is incomplete or corrupted!\")\n",
    "    print(f\"  -> Archive verified ({CROSSDOCKED_TAR.stat().st_size / 1e6:.1f} MB).\")\n",
    "\n",
    "# ------------------------------------------------------------------------------\n",
    "# 2. Download Authentic Enamine Commercial Inventory (130k Real Compounds)\n",
    "# ------------------------------------------------------------------------------\n",
    "print(\"\\n[2/3] Acquiring authentic Enamine commercial building-block inventory...\")\n",
    "if not RAW_VENDOR_CSV.exists() or RAW_VENDOR_CSV.stat().st_size < 1e6:\n",
    "    print(f\"  -> Fetching real vendor inventory from: {REAL_ENAMINE_URL}\")\n",
    "    !wget -q --show-progress \"{REAL_ENAMINE_URL}\" -O \"{RAW_VENDOR_CSV}\"\n",
    "else:\n",
    "    print(f\"  -> Authentic vendor CSV already present on disk ({RAW_VENDOR_CSV.stat().st_size / 1e6:.2f} MB).\")\n",
    "\n",
    "# Inspect raw CSV columns\n",
    "df_raw = pd.read_csv(RAW_VENDOR_CSV)\n",
    "print(f\"  -> Raw vendor records loaded: {len(df_raw):,} compounds with columns: {list(df_raw.columns)}\")\n",
    "if \"smiles\" not in df_raw.columns:\n",
    "    # Standardize column name if capitalized\n",
    "    for col in df_raw.columns:\n",
    "        if \"smi\" in col.lower():\n",
    "            df_raw.rename(columns={col: \"smiles\"}, inplace=True)\n",
    "            break\n",
    "    df_raw.to_csv(RAW_VENDOR_CSV, index=False)\n",
    "\n",
    "# ------------------------------------------------------------------------------\n",
    "# 3. Run Official Curation Script (build_synthon_catalog.py) in Strict Mode\n",
    "# ------------------------------------------------------------------------------\n",
    "print(\"\\n[3/3] Curating strictly verified 50,000+ building-block library via build_synthon_catalog.py...\")\n",
    "print(\"  -> Filtering: Heavy Atoms >= 4, 80 <= MW <= 220 Da, Fsp3 >= 0.40, Valid Reaction Handles, >= 50k unique.\")\n",
    "\n",
    "!python scripts/build_synthon_catalog.py \\\n",
    "    --input \"{RAW_VENDOR_CSV}\" \\\n",
    "    --output \"{CATALOG_PARQUET}\" \\\n",
    "    --min-fsp3 0.40 \\\n",
    "    --max-mw 220.0 \\\n",
    "    --min-heavy-atoms 4 \\\n",
    "    --strict-min-size 50000\n",
    "\n",
    "# Validate curated parquet output\n",
    "assert CATALOG_PARQUET.exists(), \"FAIL: Curated catalog was not generated!\"\n",
    "curated_df = pd.read_parquet(CATALOG_PARQUET)\n",
